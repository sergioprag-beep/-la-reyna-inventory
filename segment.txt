function findCost(name){const p=(master.products||[]).find(x=>String(x.name).toLowerCase()===String(name).toLowerCase());return p?Number(p.finalCost||p.cost||p.price||0):0}
function unitToBase(q,u){const m={g:1,oz:28.349523125,lb:453.59237,kg:1000};const v={mL:1,'fl oz':29.5735295625,L:1000,pt:473.176473,qt:946.352946,gal:3785.411784};if(m[u])return {group:'mass',v:q*m[u]};if(v[u])return {group:'vol',v:q*v[u]};return {group:'unit',v:q}}
function calculateRecipe(){let total=0;document.querySelectorAll('.recipe-row').forEach(row=>{const name=row.querySelector('.ri-product')?.value||'';const q=Number(row.querySelector('.ri-qty')?.value||0);const u=row.querySelector('.ri-unit')?.value||'lb';let c=findCost(name);let p=(master.products||[]).find(x=>String(x.name).toLowerCase()===name.toLowerCase());if(p&&p.unit&&p.unit!==u){const a=unitToBase(1,p.unit),b=unitToBase(q,u);if(a.group===b.group)c*=b.v/a.v;else c*=q}else c*=q;total+=c;const out=row.querySelector('.ri-cost');if(out)out.textContent=money(c)});const y=Number(document.getElementById('ry')?.value||0);document.getElementById('recipeCost').textContent=money(total);document.getElementById('recipeUnitCost').textContent=y?`· ${money(total/y)} por ${document.getElementById('ru').value}`:'';return total}
